enum AppMode: String, Codable {
    case normal
    case focus
    case ceo
}

final class ModeController: ObservableObject {

    static let shared = ModeController()

    @Published private(set) var currentMode: AppMode = .normal

    func enter(_ mode: AppMode) {
        currentMode = mode
        persist()
    }

    func exitCurrentMode() {
        currentMode = .normal
        persist()
    }

    private func persist() {
        UserDefaults.standard.set(currentMode.rawValue, forKey: "currentMode")
    }

    func restore() {
        if let raw = UserDefaults.standard.string(forKey: "currentMode"),
           let mode = AppMode(rawValue: raw) {
            currentMode = mode
        }
    }
}

final class EnforcementService {

    static let shared = EnforcementService()
    private let store = ManagedSettingsStore()

    func block(apps: Set<ApplicationToken>) {
        store.shield.applications = apps
    }

    func block(websites: Set<WebDomainToken>) {
        store.shield.webDomains = websites
    }

    func unblockAll() {
        store.clearAllSettings()
    }

        store.shield.applications = .none

    }
}

4️⃣ CountdownController (fiable)

import UIKit

final class CountdownController {

    static let shared = CountdownController()
    private var timer: Timer?
    private var remaining = 0
    private var isCancelled = false

    func start(seconds: Int) async -> Bool {
        remaining = seconds
        isCancelled = false

        UIApplication.shared.isIdleTimerDisabled = true

        while remaining > 0 {
            try? await Task.sleep(nanoseconds: 1_000_000_000)
            if isCancelled || UIApplication.shared.applicationState != .active {
                reset()
                return false
            }
            remaining -= 1
        }

        reset()
        return true
    }

    func cancel() {
        isCancelled = true
    }

    private func reset() {
        UIApplication.shared.isIdleTimerDisabled = false
    }
}

**5** **S’il quitte** :
	* **Streak remise à zéro**
6 Tant qu’il est en Focus :
	* Impossible de désactiver le blocage autrement

I — FocusSessionManager (source de vérité)

import Foundation

final class FocusSessionManager: ObservableObject {

    static let shared = FocusSessionManager()

    @Published private(set) var isActive: Bool = false
    private var startDate: Date?

    func start() {
        isActive = true
        startDate = Date()
        persist()
    }

    func requestExit() async -> Bool {
        let confirmed = await CountdownController.shared.start(seconds: 60)
        if confirmed {
            terminate()
            return true
        }
        return false
    }

    func terminate() {
        isActive = false
        startDate = nil
        persist()
        StreakManager.shared.reset()
        EnforcementService.shared.unblockAll()
        ModeController.shared.exitCurrentMode()
    }

    private func persist() {
        UserDefaults.standard.set(isActive, forKey: "focusActive")
        UserDefaults.standard.set(startDate, forKey: "focusStart")
    }

    func restore() {
        isActive = UserDefaults.standard.bool(forKey: "focusActive")
        startDate = UserDefaults.standard.object(forKey: "focusStart") as? Date
    }
}

# 🔒 RÈGLES CEO MODE — CONTRAT FONCTIONNEL FINAL
Quand **CEO Mode** est actif :
1 Toutes les apps sont bloquées **SAUF** :
	* Téléphone
	* Messages
	* Calendrier
	* App of a CEO
2 Fond noir / UX minimaliste (dans l’app)
3 L’utilisateur choisit **la durée de la session**
**4** **Une seule sortie possible** :
	* Aller dans App of a CEO
	* Onglet CEO Mode
	* Double confirmation
	* Message : *"Tu avais prévu de rester concentré jusqu’à 16h30. Il te reste 47 minutes."* *Quitter maintenant = streak remise à zéro* 
5 Puis **COUNTDOWN 10 MINUTES**
	* Écran noir
	* Affiche **10 → 9 → … → 0**
	* **1 chiffre = 1 minute**
	* L’utilisateur doit **rester sur l’écran**
	* Si :
		* app quittée
		* écran éteint
		* multitâche → **RESET du countdown**
6 À tout moment :
	* Il peut **annuler** et revenir en CEO Mode
7 S’il quitte :
	* Streak remise à zéro
	* Blocages levés

### I — CEOStateManager (source de vérité)

import Foundation

final class CEOSessionManager: ObservableObject {

    static let shared = CEOSessionManager()

    @Published private(set) var isActive: Bool = false
    private(set) var endDate: Date?

    func start(durationMinutes: Int) {
        isActive = true
        endDate = Calendar.current.date(byAdding: .minute, value: durationMinutes, to: Date())
        persist()
    }

    func terminate() {
        isActive = false
        endDate = nil
        persist()

        EnforcementService.shared.unblockAll()
        ModeController.shared.exitCurrentMode()
        StreakManager.shared.reset()
    }

    func remainingMinutes() -> Int {
        guard let endDate else { return 0 }
        return max(0, Int(endDate.timeIntervalSinceNow / 60))
    }

    private func persist() {
        UserDefaults.standard.set(isActive, forKey: "ceoActive")
        UserDefaults.standard.set(endDate, forKey: "ceoEndDate")
    }

    func restore() {
        isActive = UserDefaults.standard.bool(forKey: "ceoActive")
        endDate = UserDefaults.standard.object(forKey: "ceoEndDate") as? Date
    }
}

### **IV — Countdown 10 minutes STRICT**
### **Controller central**

final class CEOCountdownController: ObservableObject {

    static let shared = CEOCountdownController()

    @Published var currentMinute = 10
    private var timer: Timer?

    func start(onFinish: @escaping () -> Void) {
        currentMinute = 10
        timer?.invalidate()

        timer = Timer.scheduledTimer(withTimeInterval: 60, repeats: true) { _ in
            self.currentMinute -= 1

            if self.currentMinute <= 0 {
                self.timer?.invalidate()
                onFinish()
            }
        }
    }

    func cancel() {
        timer?.invalidate()
        currentMinute = 10
    }
}

### Vue UI

struct CEOCountdownView: View {

    @StateObject private var controller = CEOCountdownController.shared
    @Environment(\.scenePhase) var scenePhase

    var body: some View {
        ZStack {
            Color.black.ignoresSafeArea()

            Text("\(controller.currentMinute)")
                .font(.system(size: 96, weight: .bold))
                .foregroundColor(.white)
        }
        .onAppear {
            UIApplication.shared.isIdleTimerDisabled = true

            controller.start {
                CEOSessionManager.shared.terminate()
                UIApplication.shared.isIdleTimerDisabled = false
            }
        }
        .onDisappear {
            controller.cancel()
            UIApplication.shared.isIdleTimerDisabled = false
        }
        .onChange(of: scenePhase) { phase in
            if phase != .active {
                controller.cancel()
            }
        }
    }
}
