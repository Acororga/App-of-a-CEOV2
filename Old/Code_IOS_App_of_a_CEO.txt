2️⃣ Persistance sécurisée (UserDefaults chiffré simple)
final class SecureStorage {

    static let shared = SecureStorage()
    private let key = "app_limits"

    func save(_ limits: [AppLimit]) {
        let data = try? JSONEncoder().encode(limits)
        UserDefaults.standard.set(data, forKey: key)
    }

    func load() -> [AppLimit] {
        guard let data = UserDefaults.standard.data(forKey: key),
              let limits = try? JSONDecoder().decode([AppLimit].self, from: data) else {
            return []
        }
        return limits
    }
}

3️⃣ BlockingManager — cœur logique

import ManagedSettings

final class BlockingManager {

    static let shared = BlockingManager()
    private let store = ScreenTimeStore.shared

    private(set) var limits: [AppLimit] = []

    private init() {
        limits = SecureStorage.shared.load()
    }

    func applyLimits() {
        store.shield.applications = Set(
            limits.compactMap { $0.application }
                .filter { limitExceeded(for: $0) }
        )

        store.shield.webDomains = Set(
            limits.compactMap { $0.webDomain }
                .filter { limitExceeded(for: $0) }
        )
    }

    private func limitExceeded(for token: ApplicationToken) -> Bool {
        limits.first(where: { $0.application == token })?.usedMinutesToday ?? 0
        >= limits.first(where: { $0.application == token })?.dailyLimitMinutes ?? 0
    }

    private func limitExceeded(for domain: WebDomain) -> Bool {
        limits.first(where: { $0.webDomain == domain })?.usedMinutesToday ?? 0
        >= limits.first(where: { $0.webDomain == domain })?.dailyLimitMinutes ?? 0
    }
}

import SwiftUI

struct ConfirmationCountdownView: View {

    let duration: Int
    let onConfirm: () -> Void
    let onCancel: () -> Void

    @State private var remaining: Int
    @State private var isCompleted = false

    init(duration: Int,
         onConfirm: @escaping () -> Void,
         onCancel: @escaping () -> Void) {
        self.duration = duration
        self.onConfirm = onConfirm
        self.onCancel = onCancel
        _remaining = State(initialValue: duration)
    }

    var body: some View {
        VStack(spacing: 32) {
            Text("\(remaining)")
                .font(.system(size: 64, weight: .bold))

            if isCompleted {
                Button("Valider") {
                    onConfirm()
                }
            } else {
                Button("Annuler") {
                    onCancel()
                }
            }
        }
        .onAppear {
            Timer.scheduledTimer(withTimeInterval: 1, repeats: true) { timer in
                remaining -= 1
                if remaining <= 0 {
                    timer.invalidate()
                    isCompleted = true
                }
            }
        }
    }
}

import ManagedSettings
import FamilyControls

final class FocusManager {

    static let shared = FocusManager()
    private let store = ScreenTimeStore.shared
    private let storageKey = "focus_session"

    private init() {}

    // MARK: - Session State

    var currentSession: FocusSession? {
        guard let data = UserDefaults.standard.data(forKey: storageKey),
              let session = try? JSONDecoder().decode(FocusSession.self, from: data),
              session.isActive else {
            return nil
        }
        return session
    }

    // MARK: - Start Focus

    func startFocus(durationMinutes: Int) {
        let session = FocusSession(
            startDate: Date(),
            plannedEndDate: Date().addingTimeInterval(TimeInterval(durationMinutes * 60)),
            isActive: true
        )

        save(session)
        applyFocusBlocking()
    }

    // MARK: - End Focus

    func endFocus() {
        clearSession()
        removeFocusBlocking()
        StreakManager.shared.reset()
    }

    // MARK: - Persistence

    private func save(_ session: FocusSession) {
        let data = try? JSONEncoder().encode(session)
        UserDefaults.standard.set(data, forKey: storageKey)
    }

    private func clearSession() {
        UserDefaults.standard.removeObject(forKey: storageKey)
    }
}

extension FocusManager {

    func applyFocusBlocking() {
        let limits = BlockingManager.shared.limits

        store.shield.applications = Set(
            limits.compactMap { $0.application }
        )

        store.shield.webDomains = Set(
            limits.compactMap { $0.webDomain }
        )
    }

    func removeFocusBlocking() {
        store.shield.applications = nil
        store.shield.webDomains = nil

        // Réappliquer le blocage classique après focus
        BlockingManager.shared.applyLimits()
    }
}

## 6️⃣ Countdown 10 minutes — ÉCRAN BLOQUANT
### 🔒 Contraintes techniques critiques :
* UIApplication.shared.isIdleTimerDisabled = true
* reset si :
  * app passe en background
  * écran s’éteint
  * changement de view

struct CEOExitCountdownView: View {

    @State private var remaining = 10
    @State private var timer: Timer?

    let onSuccess: () -> Void
    let onCancel: () -> Void

    var body: some View {
        ZStack {
            Color.black.ignoresSafeArea()

            VStack {
                Text("\(remaining)")
                    .foregroundColor(.white)
                    .font(.system(size: 96, weight: .bold))

                Button("Annuler") {
                    cancel()
                }
                .foregroundColor(.white)
            }
        }
        .onAppear {
            UIApplication.shared.isIdleTimerDisabled = true
            start()
        }
        .onDisappear {
            reset()
        }
        .onReceive(NotificationCenter.default.publisher(for: UIApplication.willResignActiveNotification)) { _ in
            reset()
        }
    }

    private func start() {
        remaining = 10
        timer = Timer.scheduledTimer(withTimeInterval: 60, repeats: true) { _ in
            remaining -= 1
            if remaining == 0 {
                complete()
            }
        }
    }

    private func reset() {
        timer?.invalidate()
        timer = nil
        UIApplication.shared.isIdleTimerDisabled = false
        remaining = 10
    }

    private func cancel() {
        reset()
        onCancel()
    }

    private func complete() {
        reset()
        onSuccess()
    }
}
