object CEOModeManager {

    private var endTime: Long = 0L
    var isActive = false
        private set

    fun start(context: Context, durationMinutes: Int) {
        isActive = true
        endTime = System.currentTimeMillis() + durationMinutes * 60_000

        SessionStore.saveCEO(endTime)

        AppBlockerService.start(context, whitelist = CEOWhitelist.apps)
        MinimalistOverlay.show(context)
    }

    fun terminate(context: Context) {
        isActive = false
        endTime = 0L

        SessionStore.clearCEO()
        AppBlockerService.stop(context)
        MinimalistOverlay.hide()
        StreakManager.reset()
    }

    fun remainingMinutes(): Int {
        return ((endTime - System.currentTimeMillis()) / 60_000).toInt().coerceAtLeast(0)
    }
}

object MinimalistOverlay {

        if (view != null) return

            TYPE_APPLICATION_OVERLAY,
            FLAG_NOT_FOCUSABLE,

        ))
    }

    fun hide() {
        view?.let {
            val wm = it.context.getSystemService(Context.WINDOW_SERVICE) as WindowManager
            wm.removeView(it)
        }
        view = null
    }
}

class CEOExitCountdownService : Service() {

    private var minutesLeft = 10
    private lateinit var wakeLock: PowerManager.WakeLock

    override fun onCreate() {
        val pm = getSystemService(POWER_SERVICE) as PowerManager
        wakeLock = pm.newWakeLock(
            PowerManager.SCREEN_BRIGHT_WAKE_LOCK,
            "CEO:Countdown"
        )
        wakeLock.acquire(10 * 60_000L)
    }

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        startCountdown()
        return START_STICKY
    }

    private fun startCountdown() {
        Handler(Looper.getMainLooper()).postDelayed(object : Runnable {
            override fun run() {
                minutesLeft--
                CountdownOverlay.update(minutesLeft)

                if (minutesLeft <= 0) {
                    CEOModeManager.terminate(this@CEOExitCountdownService)
                    stopSelf()
                } else {
                    Handler(Looper.getMainLooper()).postDelayed(this, 60_000)
                }
            }
        }, 60_000)
    }

    override fun onDestroy() {
        wakeLock.release()
        super.onDestroy()
    }

    override fun onBind(intent: Intent?): IBinder? = null
}

object NormalQuotaManager {

    fun isQuotaExceeded(packageName: String): Boolean {
        val used = UsageTracker.minutesUsedToday(packageName)
        val allowed = NormalSessionStore.getQuota(packageName)
        return used >= allowed
    }

    fun addTime(packageName: String, minutes: Int) {
        require(minutes % 10 == 0)
        NormalSessionStore.incrementQuota(packageName, minutes)
    }

    fun resetDaily() {
        NormalSessionStore.resetUsage()
    }
}

### **2️⃣ UsageTracker.kt**

object UsageTracker {

    fun minutesUsedToday(packageName: String): Int {
        val usm = App.context.getSystemService(Context.USAGE_STATS_SERVICE) as UsageStatsManager
        val start = LocalDate.now().atStartOfDay(ZoneId.systemDefault()).toEpochSecond() * 1000

        val stats = usm.queryUsageStats(
            UsageStatsManager.INTERVAL_DAILY,
            start,
            System.currentTimeMillis()
        )

        val usage = stats.firstOrNull { it.packageName == packageName }
        return ((usage?.totalTimeInForeground ?: 0L) / 60_000).toInt()
    }
}

object FocusSessionStore {

    private val prefs =
        App.context.getSharedPreferences("focus", Context.MODE_PRIVATE)

    fun start(durationMinutes: Int) {
        prefs.edit()
            .putBoolean("active", true)
            .putLong("endTime", System.currentTimeMillis() + durationMinutes * 60_000)
            .apply()
    }

    fun end() {
        prefs.edit().clear().apply()
    }

    fun isActive(): Boolean =
        prefs.getBoolean("active", false)

    fun remainingMillis(): Long =
        prefs.getLong("endTime", 0) - System.currentTimeMillis()
}

class FocusCountdownService : Service() {

    private var secondsLeft = 60
    private lateinit var handler: Handler

    companion object {
        fun start() {
            val intent = Intent(App.context, FocusCountdownService::class.java)
            App.context.startService(intent)
        }
    }

    override fun onCreate() {
        handler = Handler(Looper.getMainLooper())
        CountdownOverlay.show("Quitter Focus ?", secondsLeft)
        tick()
    }

    private fun tick() {
        handler.postDelayed(object : Runnable {
            override fun run() {
                secondsLeft--
                CountdownOverlay.update(secondsLeft)

                if (secondsLeft <= 0) {
                    FocusSessionManager.forceExit()
                    stopSelf()
                } else {
                    handler.postDelayed(this, 1000)
                }
            }
        }, 1000)
    }

    override fun onDestroy() {
        handler.removeCallbacksAndMessages(null)
    }

    override fun onBind(intent: Intent?): IBinder? = null
}

object StreakManager {

    private val prefs =
        App.context.getSharedPreferences("streak", Context.MODE_PRIVATE)

    fun resetStreak() {
        prefs.edit()
            .putInt("currentStreak", 0)
            .putLong("lastBrokenAt", System.currentTimeMillis())
            .apply()
    }

    fun incrementStreak() {
        val current = prefs.getInt("currentStreak", 0)
        prefs.edit()
            .putInt("currentStreak", current + 1)
            .apply()
    }

    fun getCurrentStreak(): Int =
        prefs.getInt("currentStreak", 0)
}

### 3️⃣ CeoModeManager.kt — CŒUR DU SYSTÈME

object CeoModeManager {

    private val dpm =
        App.context.getSystemService(Context.DEVICE_POLICY_SERVICE)
                as DevicePolicyManager

    private val admin = ComponentName(
        App.context,
        AdminReceiver::class.java
    )

    private val allowedPackages = listOf(
        "com.android.dialer",
        "com.android.messaging",
        "com.android.calendar",
        App.context.packageName
    )

    fun enable(durationMinutes: Int) {
        lockDevice()
        restrictApps()
        setImmersive()
        scheduleEnd(durationMinutes)
        CeoState.activate(durationMinutes)
    }

    fun disable() {
        unlockDevice()
        restoreApps()
        CeoState.deactivate()
    }
}
